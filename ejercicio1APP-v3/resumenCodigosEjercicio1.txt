funcionalidades:
    El sistema es una aplicacion de escritorio movil que integra multiples
    herramientas de calculo y aprendizaje. 
    Incluye un Menu Principal (MainActivity) que permite navegar entre 
    las siguientes funciones: 
        1. Suma de dos numeros con visualizacion de operandos y resultado.
        2. Conversion de moneda de Pesos Chilenos a Dolares (CLP a USD) con validacion de valor del dolar. 
        3. Calculo de cuotas en pesos a partir de valores en UF con normalizacion inteligente de separadores de miles y decimales. 
        4. Calculo del Indice de Masa Corporal (IMC) con diagnostico basado en rangos de la OMS. 
        5. Calculo de Sueldo Anual a partir del sueldo mensual. 
        6. Conversion de unidades de longitud de Centimetros a Pies. 
        7. Demostracion de estructuras de control repetitivas (Ciclos While, Do-While y For) contando repeticiones. 
        8. Calculo de faltas permitidas para mantener un 70% de asistencia en un curso de 12 clases. 
        9. Calculo de Sueldo Neto que incluye un sueldo base de $300.000 y un bono del 12% por ventas, 
            permitiendo el ingreso de 3 ventas estaticas o la generacion dinamica de N campos de entrada 
            en una pantalla secundaria (DetalleVentas). 
            El codigo incluye validaciones de entrada, manejo de excepciones de conversion, 
            formateo de decimales y uso de Intents para navegacion entre actividades, 
            manteniendo comentarios explicativos y didacticos para cada bloque logico.

                    if (valorDolar > 0.0) {

                        txvValorDolar.text=valorDolar.toString()
                        txvPesos.text=pesos.toString()
                        // Formateamos a 2 decimales para una presentacion profesional
                        txvDolares.text= String.format("%.2f", totalDolares)
                        val mensaje= "Con el valor del dolar a ${valorDolar.toString()} y con ${pesos.toString()} Pesos Tiene ${totalDolares.toString()} dolares"
                        Toast.makeText(this,mensaje, Toast.LENGTH_LONG).show()
                    }
                }
            }

    import java.text.DecimalFormat
    import java.text.DecimalFormatSymbols
    import java.util.Locale
    class CuotaEnPesos : AppCompatActivity() {
        override fun onCreate(savedInstanceState: Bundle?) {
            super.onCreate(savedInstanceState)
            enableEdgeToEdge()
            setContentView(R.layout.activity_cuota_en_pesos)
            ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
                val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
                v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
                insets
            }
            val btnCalculo = findViewById<Button>(R.id.btnCalculoCuota)
            val btnMenu = findViewById<Button>(R.id.btnMenu)
            val txtInputValUf = findViewById<EditText>(R.id.txtInputValorUf)
            val txtInputCosto = findViewById<EditText>(R.id.txtInputCostoCred)
            val txvOutCuota = findViewById<TextView>(R.id.txvCuota)
            btnCalculo.setOnClickListener {
                val strUf = txtInputValUf.text.toString().trim()
                val strCosto = txtInputCosto.text.toString().trim()
                // Llama a la funcion de normalizacion para limpiar los caracteres no numericos (puntos y comas)
                val valUf = normalizarMonto(strUf)
                val costoUf = normalizarMonto(strCosto)
                if (valUf <= 0.0 || costoUf <= 0.0) {
                    Toast.makeText(this, "Por favor, ingrese valores válidos", Toast.LENGTH_SHORT).show()
                    return@setOnClickListener
                }
                val resultado = valUf * costoUf
                // Formateo de Salida: Usa formato de moneda con puntos para miles (Estilo Chile)
                val symbols = DecimalFormatSymbols(Locale.GERMAN)
                val formatter = DecimalFormat("$ #,###", symbols)
                val cuotaFormateada = formatter.format(Math.round(resultado))
                txvOutCuota.text = cuotaFormateada
                Toast.makeText(this, "Calc: $valUf * $costoUf", Toast.LENGTH_SHORT).show()
                txtInputCosto.text.clear()
                txtInputValUf.text.clear()
                txtInputValUf.requestFocus()
            }
            btnMenu.setOnClickListener { finish() }
        }
        // Funcion didactica: Maneja la inconsistencia de los usuarios al ingresar decimales o miles
        private fun normalizarMonto(monto: String): Double {
            if (monto.isEmpty()) return 0.0
            return try {
                var temp = monto.trim()
                // Si el texto tiene ambos (punto y coma), asume formato europeo/chileno
                if (temp.contains(".") && temp.contains(",")) {
                    temp = temp.replace(".", "").replace(",", ".")
                }
                else if (temp.contains(",")) {
                    temp = temp.replace(",", ".")
                }
                else if (temp.contains(".")) {
                    val partes = temp.split(".")
                    // Si despues del punto hay 3 cifras, probablemente sea separador de miles
                    if (partes.size > 1 && partes.last().length == 3) {
                        temp = temp.replace(".", "")
                    }
                }
                temp.toDoubleOrNull() ?: 0.0
            } catch (e: Exception) { 0.0 }
        }
    }

        override fun onCreate(savedInstanceState: Bundle?) {
            super.onCreate(savedInstanceState)
            enableEdgeToEdge()
            setContentView(R.layout.activity_imc)
            ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
                val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
                v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
                insets
            }
            val btnMenu = findViewById<Button>(R.id.btnMenu)
            val btnCalculo = findViewById<Button>(R.id.btnCalcularIMC)

            btnCalculo.setOnClickListener {
                // Reemplazamos la coma por punto para asegurar compatibilidad con Double.toDouble()
                val strEstatura = txtInputEstatura.text.toString().trim().replace(",", ".")
                val strPeso = txtInputPeso.text.toString().trim().replace(",", ".")
                val estatura = strEstatura.toDoubleOrNull() ?: 0.0
                val peso = strPeso.toDoubleOrNull() ?: 0.0
                // Validaciones de rango logico
                if (estatura <= 0.0 || estatura > 2.5) {
                    Toast.makeText(this, "Ingrese una estatura válida (ej: 1.75)", Toast.LENGTH_SHORT).show()
                    txtInputEstatura.requestFocus()
                    return@setOnClickListener
                }
                if (peso <= 0.0) {
                    Toast.makeText(this, "Ingrese un peso válido", Toast.LENGTH_SHORT).show()
                    txtInputPeso.requestFocus()
                    return@setOnClickListener
                }
                // Formula IMC = peso / estatura al cuadrado
                val imc = peso / (estatura * estatura)
                // Estructura condicional para determinar el estado nutricional
                val diagnostico = when {
                    imc < 18.5 -> "Bajo peso"
                    imc < 25.0 -> "Peso normal"
                    imc < 30.0 -> "Sobrepeso"
                    else -> "Obesidad"
                }
                val mensaje = String.format("Tu IMC es: %.2f (%s)", imc, diagnostico)
                txvOutImc.text = mensaje
                Toast.makeText(this, mensaje, Toast.LENGTH_SHORT).show()
                txtInputEstatura.text.clear()
                txtInputPeso.text.clear()
                txtInputPeso.requestFocus()
            }
            btnMenu.setOnClickListener { finish() }
        }
    }

        // Funcion que usa While: Se comprueba la condicion ANTES de entrar
        fun cicloWhile10():Int{
            var contador = 0
            var repeticiones=0
            while(contador<=10){
                repeticiones++
                contador++
            }
            return repeticiones
        }
        // Funcion que usa Do-While: El bloque se ejecuta AL MENOS UNA VEZ antes de comprobar
        fun cicloDoWhile10():Int{
            var contador=0
            var repeticiones=0
            do {
                repeticiones++
                contador++
        } while(contador<=10)
        return repeticiones
        }
        // Funcion que usa For: Itera sobre un rango definido (de 0 a 10 inclusive)
        fun cicloFor10():Int{
            var repeticiones=0
            for (i in 0..10) {
                repeticiones++
            }
            return repeticiones
        }
    }

    package com.example.ejercicio1app
    import android.os.Bundle
    import android.text.InputType
    import android.widget.Button

    import androidx.appcompat.app.AlertDialog
    import androidx.appcompat.app.AppCompatActivity
    import androidx.core.view.ViewCompat
    import androidx.core.view.WindowInsetsCompat
    class DetalleVentas : AppCompatActivity() {
        private val sueldoBase = 300000.0
        private val bonoPorVenta = 0.12
        // Lista dinamica para rastrear los campos creados en tiempo de ejecucion
        private val listaInputs = mutableListOf<EditText>()
        override fun onCreate(savedInstanceState: Bundle?) {
            super.onCreate(savedInstanceState)
            enableEdgeToEdge()
            setContentView(R.layout.activity_detalle_ventas)
            ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
                val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
                v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
                insets
            }
            val container = findViewById<LinearLayout>(R.id.containerVentas)
            val btnCalcular = findViewById<Button>(R.id.btnCalcularFinal)
            val btnVolver = findViewById<Button>(R.id.btnVolver)
            btnVolver.setOnClickListener { finish() }
            // Recibe el dato enviado por la pantalla anterior
            val cantVentas = intent.getIntExtra("CANTIDAD_VENTAS", 0)
            // Bucle didactico: Crea elementos visuales programaticamente (programacion de interfaz dinamica)
            for (i in 1..cantVentas) {
                val input = EditText(this).apply {
                    hint = "Monto Venta $i ($)"
                    inputType = InputType.TYPE_CLASS_NUMBER or InputType.TYPE_NUMBER_FLAG_DECIMAL
                    layoutParams = LinearLayout.LayoutParams(
                        LinearLayout.LayoutParams.MATCH_PARENT,
                        LinearLayout.LayoutParams.WRAP_CONTENT
                    ).apply { setMargins(0, 0, 0, 16) }
                }
                listaInputs.add(input) // Guardamos la referencia para sumar despues
                container.addView(input) // Inyectamos el componente al contenedor vertical (LinearLayout)
            }
            btnCalcular.setOnClickListener {
                var totalMontoVentas = 0.0
                // Iteramos sobre la lista de referencias para obtener los valores
                for (input in listaInputs) {
                    val monto = input.text.toString().toDoubleOrNull() ?: 0.0
                    totalMontoVentas += monto
                }
                val totalBono = totalMontoVentas * bonoPorVenta
                val sueldoNeto = sueldoBase + totalBono
                // Presentacion de resultados mediante un Dialogo de Alerta
                AlertDialog.Builder(this)
                    .setTitle("Resumen Sueldo Neto")
                    .setMessage("Sueldo Base: $$sueldoBase\n" +
                                "Ventas Ingresadas ($cantVentas): $$totalMontoVentas\n" +
                                "Bono (12%): $$totalBono\n" +
                                "--------------------------------\n" +
                                "SUELDO NETO: $$sueldoNeto")
                    .setPositiveButton("Aceptar", null)
                    .show()
            }
        }
    }
